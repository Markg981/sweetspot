# =====================================================================
#  SWEETSPOT - impostazioni
# =====================================================================
#  Questo file si apre e si modifica con il Blocco note, da qualunque
#  computer. Salva, rimetti la chiavetta nel player e riaccendilo.
#
#  Non serve cambiare niente: i valori qui sotto sono quelli consigliati.
#  Le righe che iniziano con # sono commenti e vengono ignorate.
# =====================================================================

# La maggior parte delle impostazioni si cambia anche dal browser:
# http://sweetspot.local:8080/ (pagine Musica e Impostazioni).

# Nome del player, come apparira' nelle app e in rete.
NOME_PLAYER=Sweetspot

# Modalita' di funzionamento:
#   completa = libreria, server e interfaccia web su Sweetspot (consigliato).
#              Dal telefono o dal computer apri http://sweetspot.local/
#   lyrion   = solo player, per un Lyrion Music Server su un altro computer
MODALITA=completa

# --- Musica ---------------------------------------------------------------
# Dischi interni e USB vengono trovati da soli e letti in sola lettura.
# DISCHI=no li ignora tutti; ESCLUDI_DISCHI ignora quelli indicati per nome,
# separati da virgole (es. ESCLUDI_DISCHI=Backup,Dati vecchi).
DISCHI=si
ESCLUDI_DISCHI=

# Cartelle condivise in rete (NAS, PC Windows, Mac), fino a 9.
# Esempio:
#   CONDIVISIONE_1=//192.168.1.10/Musica
#   CONDIVISIONE_1_UTENTE=marco
#   CONDIVISIONE_1_PASSWORD=la-password
# Senza utente si accede come ospite. Si aggiungono piu' comodamente dalla
# pagina Musica.

# --- Solo modalita' lyrion ------------------------------------------------
# Indirizzo del server Lyrion Music Server.
# Vuoto = Sweetspot lo cerca da solo nella rete di casa.
SERVER=

# Quale DAC usare.
# auto = il primo DAC USB collegato (consigliato).
# Puoi anche scrivere il nome, per esempio DAC=R26.
DAC=auto

# Profilo di funzionamento del processore:
#   bilanciato  = frequenza fissa intermedia (consigliato)
#   silenzio    = frequenza fissa al minimo: meno consumo e meno calore
#   prestazioni = frequenza fissa alla nominale, processore sempre sveglio
PROFILO=bilanciato

# DSD:
#   auto   = DSD nativo se il DAC lo dichiara, altrimenti convertito in PCM
#   nativo = forza il DSD nativo (per DAC che lo supportano ma non lo dichiarano)
#   dop    = DSD tramite DoP. Solo se il DAC lo supporta: altrimenti si sente
#            un forte fruscio!
#   no     = il DSD viene sempre convertito in PCM
DSD=auto

# Pausa (in millisecondi) quando si passa da PCM a DSD, per dare tempo
# al DAC di agganciarsi senza perdere l'inizio del brano (modalita' lyrion).
PAUSA_DSD_MS=500

# Pausa (in millisecondi) a ogni cambio di frequenza di campionamento
# (modalita' lyrion). 0 = nessuna pausa. Prova 300-500 se senti un clic.
PAUSA_FREQUENZA_MS=0

# Rete Wi-Fi, solo se il player non e' collegato con il cavo.
# Il cavo e' sempre preferibile.
WIFI_NOME=
WIFI_PASSWORD=

# Risparmio energetico della scheda di rete (EEE). no = spento (consigliato).
RISPARMIO_RETE=no

# Usare l'uscita audio integrata del computer al posto del DAC USB.
USCITA_INTEGRATA=no

# Dopo quanti minuti spegnere lo schermo (0 = mai).
SCHERMO_MINUTI=2

# Accesso remoto per l'assistenza tecnica (SSH).
SSH=no
SSH_PASSWORD=

# Ottimizzazioni di sistema (core dedicati, frequenza fissa, ecc.).
# no = funzionamento normale, utile solo per confronti o problemi.
OTTIMIZZAZIONI=si
