# =====================================================================
#  SWEETSPOT - impostazioni
# =====================================================================
#  Non serve aprire questo file: tutte le impostazioni si cambiano dal
#  browser, dal telefono o dal computer, su http://sweetspot.local/
#  (menu dell'interfaccia > Impostazioni Sweetspot).
#
#  Se preferisci, si modifica anche con il Blocco note: salva, rimetti la
#  chiavetta nel player e riaccendilo. Le righe che iniziano con # sono
#  commenti. I valori qui sotto sono quelli consigliati.
# =====================================================================

# Nome del player, come appare nell'interfaccia e nelle app.
NOME_PLAYER=Sweetspot

# Modalita' di funzionamento:
#   completa = come Daphile: libreria, plugin (Qobuz, TIDAL, Spotify,
#              radio...) e interfaccia su Sweetspot (consigliato).
#   player   = solo player, per un Lyrion Music Server su un altro computer.
MODALITA=completa

# Solo modalita' player: indirizzo del server Lyrion.
# Vuoto = Sweetspot lo cerca da solo nella rete di casa.
SERVER=

# Modalita' ascolto: mentre la musica suona, la lettura della libreria si
# sospende e riparte quando la musica si ferma.
MODALITA_ASCOLTO=si

# Precarico in RAM (solo modalita' completa): i brani della coda, dal brano
# in corso alla fine, vengono letti e tenuti in RAM; quando c'e' tutta la
# coda i dischi meccanici che la contengono vanno in standby. Lo streaming
# non si precarica. PRECARICO_MB limita la memoria usata (auto = meta' di
# quella libera); i file piu' grandi di PRECARICO_FILE_MAX_MB si leggono dal
# disco; PRECARICO_STANDBY=no lascia i dischi accesi.
PRECARICO=si
PRECARICO_MB=auto
PRECARICO_FILE_MAX_MB=1024
PRECARICO_STANDBY=si

# Tetti di memoria (modalita' completa). LYRION_MEM_MB e SCANNER_MEM_MB:
# limite cgroup di Lyrion e del suo scanner (auto = 70% e 40% della RAM);
# ARTWORK_RAM_MB: cache delle copertine in RAM quando non c'e' l'archivio
# musicale (auto = 4% della RAM, tra 32 e 256 MB).
LYRION_MEM_MB=auto
SCANNER_MEM_MB=auto
ARTWORK_RAM_MB=auto

# Cache delle copertine di Lyrion: CACHE_COPERTINE = auto (sull'archivio se c'e',
# se no in RAM), ram, archivio oppure disco (un altro disco scrivibile, per
# nome: CACHE_COPERTINE_DISCO=Nome). CACHE_COPERTINE_MB: tetto sul disco
# (auto = 1024 MB). Se il disco manca si ripiega in RAM e la pagina Stato lo dice.
CACHE_COPERTINE=auto
CACHE_COPERTINE_DISCO=
CACHE_COPERTINE_MB=auto

# --- Musica ---------------------------------------------------------------
# Dischi interni e USB vengono trovati da soli e letti in sola lettura.
# DISCHI=no li ignora tutti; ESCLUDI_DISCHI ignora quelli indicati per nome,
# separati da virgole (es. ESCLUDI_DISCHI=Backup,Dati vecchi).
DISCHI=si
ESCLUDI_DISCHI=

# Cartelle condivise in rete (NAS, PC Windows, Mac), fino a 9. Si
# aggiungono piu' comodamente dalla pagina Musica. Esempio:
#   CONDIVISIONE_1=//192.168.1.10/Musica
#   CONDIVISIONE_1_UTENTE=marco
#   CONDIVISIONE_1_PASSWORD=la-password
#   CONDIVISIONE_1_SMB1=si          (solo per NAS molto vecchi: SMB1 e' insicuro)
#   CONDIVISIONE_2=nfs://192.168.1.10/export/musica   (cartella NFS)
# Senza utente si accede come ospite.

# --- Audio ----------------------------------------------------------------
# Quale DAC usare: auto = il primo DAC USB collegato (consigliato; con le
# zone qui sotto resta lui per tutto l'avvio). Puoi anche scrivere il nome,
# per esempio DAC=R26, o l'identificativo USB (292b:0a26): nella pagina
# Audio si sceglie tra i DAC collegati.
DAC=auto

# Altri DAC USB: no = suona solo il DAC qui sopra; si = ogni altro DAC
# collegato diventa un player separato in Lyrion (zone da raggruppare li').
ALTRI_DAC=no

# Volume:
#   fisso    = 100%, si regola dall'amplificatore (bit-perfect, consigliato)
#   dac      = usa il controllo di volume del DAC (bit-perfect)
#   software = attenuazione digitale di Squeezelite, con dither TPDF
#   dsp      = volume a 64 bit con dither (CamillaDSP), solo in modalita'
#              completa: il migliore se il volume va regolato da Lyrion
VOLUME=fisso

# DSD:
#   auto   = DSD nativo se il DAC lo dichiara, altrimenti convertito in PCM da Squeezelite
#   nativo = forza il DSD nativo (per DAC che lo supportano senza dichiararlo)
#   dop    = DSD tramite DoP. Solo se il DAC lo supporta: altrimenti si sente
#            un forte fruscio!
#   no     = il DSD viene sempre convertito in PCM
DSD=auto
# Margine (dB) quando Squeezelite converte il DSD in PCM: da -24 a 0, di serie -3.
# I DSD "over" superano 0 dBFS fino a +3 dB (il 50% di modulazione oltre lo
# standard): senza margine la conversione li taglierebbe. Il prezzo e' che tutto
# il DSD convertito suona 3 dB piu' piano (il volume del DAC lo compensa).
# 0 = nessun margine, solo se i tuoi file non superano 0 dBFS.
DSD_PCM_MARGINE_DB=-3

# Ricampionamento (cambia il segnale, quindi di serie e' spento):
#   no        = ogni brano alla sua frequenza originale (bit-perfect)
#   sincrono  = multiplo intero: 44,1 kHz verso 176,4/352,8 e 48 verso 192/384
#   asincrono = tutto alla frequenza massima del DAC
# FILTRO: lineare (predefinito), intermedio, minimo (fase minima).
RICAMPIONAMENTO=no
FILTRO=lineare

# Correzione ambientale con i filtri misurati con REW (pagina Audio >
# Correzione ambientale). Cambia il segnale, quindi di serie e' spenta.
# I filtri sono nella cartella correzione-ambientale (testo esportato da REW:
# sinistro.txt e destro.txt). Nella stessa cartella si possono copiare dal
# computer anche impulsi FIR in WAV (rePhase, Acourate, REW): fir-sinistro.wav
# e, se diverso, fir-destro.wav; CamillaDSP lavora alla loro frequenza.
CORREZIONE=no
# Attenuazione in piu' (dB) quando ci sono i filtri FIR.
FIR_MARGINE_DB=3

# Frequenza massima inviata al DAC, in Hz. 0 = quella dichiarata dal DAC.
FREQUENZA_MAX=0

# Pausa (in millisecondi) quando si passa da PCM a DSD, per dare tempo
# al DAC di agganciarsi senza perdere l'inizio del brano.
PAUSA_DSD_MS=500

# Pausa (in millisecondi) a ogni cambio di frequenza di campionamento.
# 0 = nessuna pausa. Prova 300-500 se il DAC perde l'inizio dei brani.
PAUSA_FREQUENZA_MS=0

# Usare l'uscita audio integrata del computer al posto del DAC USB.
USCITA_INTEGRATA=no

# Solo Raspberry Pi: scheda DAC montata sul Pi (I2S), con il nome del suo
# overlay, per esempio hifiberry-dacplus, iqaudio-dacplus, allo-boss-dac-
# pcm512x-audio. Vuoto = DAC USB. Si sceglie anche dalla pagina Audio.
SCHEDA_I2S=

# --- Rete -----------------------------------------------------------------
# Nome in rete: il player risponde su http://NOME.local/
# Diverso per ogni Sweetspot della casa.
NOME_RETE=sweetspot

# Rete Wi-Fi, solo se il player non e' collegato con il cavo.
# Il cavo e' sempre preferibile.
WIFI_NOME=
WIFI_PASSWORD=

# Hotspot di configurazione (si/no): se dopo HOTSPOT_ATTESA secondi non c'e'
# nessuna connessione (niente cavo, Wi-Fi non configurato o non collegato)
# e c'e' un adattatore Wi-Fi che lo permette, il player accende la rete
# "Sweetspot-XXXX" per scegliere il Wi-Fi di casa dal telefono. La password
# compare sullo schermo (con il codice QR) e viene scritta, insieme al
# codice QR, nei file sweetspot-hotspot.txt e sweetspot-hotspot.svg di questa
# chiavetta; per sceglierla tu scrivine una (da 8 a 63 caratteri) in
# HOTSPOT_PASSWORD.
# HOTSPOT_PROVA: secondi concessi alla rete scelta prima che l'hotspot riparta.
HOTSPOT=si
HOTSPOT_ATTESA=60
HOTSPOT_PROVA=40
HOTSPOT_PASSWORD=

# Indirizzo fisso (vuoto = automatico dal router, consigliato). Esempio:
#   IP_FISSO=192.168.1.50/24
#   GATEWAY=192.168.1.1
#   DNS=192.168.1.1
IP_FISSO=

# Risparmio energetico della scheda di rete (EEE). no = spento (consigliato).
RISPARMIO_RETE=no

# --- Sistema --------------------------------------------------------------
# Profilo di funzionamento del processore:
#   bilanciato  = frequenza fissa intermedia, non dinamica (consigliato)
#   dinamico    = frequenza che segue il carico, stati di riposo profondi
#                 (esclusi i core audio isolati): meno consumo a riposo
#   silenzio    = frequenza fissa al minimo: meno consumo e meno calore
#   prestazioni = frequenza fissa alla nominale, processore sempre sveglio
PROFILO=bilanciato

# Dopo quanti minuti spegnere lo schermo (0 = mai).
SCHERMO_MINUTI=2

# Periferiche che il player non usa, su qualunque computer: webcam,
# Bluetooth e lettori di smart card USB scollegati dai driver (anche se
# collegati dopo); scheda video secondaria, audio integrato non scelto,
# modem, FireWire, lettore di schede, Wi-Fi non configurato e schede di
# rete senza cavo lasciati a riposo. accese = lasciate come sono.
# DAC, dischi, tastiere e la rete in uso non vengono mai toccati.
PERIFERICHE_INUTILIZZATE=spente

# Accesso remoto per l'assistenza tecnica (SSH).
SSH=no
SSH_PASSWORD=
# Utente del login SSH: root (predefinito) oppure assistenza, senza privilegi.
# Dalle pagine, SSH, utente e password si cambiano solo riscrivendo la
# password delle pagine (PASSWORD_WEB): aprono il sistema come amministratore.
SSH_UTENTE=root

# Connessione protetta (HTTPS) delle pagine, con certificato autofirmato
# creato al primo avvio e salvato su questa chiavetta (sweetspot-https.pem).
# HTTPS_REINDIRIZZA=si fa rimandare a https chi apre le pagine in chiaro.
HTTPS=no
HTTPS_REINDIRIZZA=no

# Renderer UPnP/DLNA in entrata: Sweetspot compare come diffusore in
# BubbleUPnP, JPlay, mconnect e nelle altre app UPnP/DLNA. L'audio passa da
# Lyrion e dal player (come la libreria), il DAC lo apre solo il player.
# Solo con MODALITA=completa. Si accende anche dalla pagina Sistema.
RENDERER_UPNP=no

# Copia dei CD: servizi online interrogati, sempre in HTTPS.
# CD_CERCA_TITOLI: titoli e copertina da MusicBrainz e Cover Art Archive
# (ricevono l'identificativo del disco, che rivela cosa si copia).
# CD_ACCURATERIP: verifica delle tracce con AccurateRip (riceve l'indice del
# disco). Con no la copia funziona lo stesso: titoli generici, senza
# copertina e/o senza verifica.
CD_CERCA_TITOLI=si
CD_ACCURATERIP=si

# Media server UPnP/DLNA: la libreria compare come archivio negli streamer e
# nei DAC di rete (es. Gustard, guidati da BubbleUPnP o mconnect), che leggono
# i file da soli. I file partono senza conversioni: DSF e DFF arrivano in DSD,
# alla frequenza del file, se il renderer li accetta. Il PC puo' stare in
# un'altra stanza. Solo con MODALITA=completa. Si accende anche dalla pagina
# Sistema.
SERVER_UPNP=no

# Password delle pagine di Sweetspot (utente admin). Obbligatoria: se e' vuota,
# al prossimo avvio il dispositivo mostra un codice univoco (console e file
# PRIMO-ACCESSO.txt su questa chiavetta) con cui scegliere la password dalla
# pagina Primo accesso. Dimenticata: togli la riga e riavvia (stesso percorso).
PASSWORD_WEB=

# Credenziali di Lyrion (porta 9000, Material, app e CLI 9090): se vuote, al
# primo avvio Sweetspot genera una password casuale (utente sweetspot) e la
# scrive qui; la pagina Sistema la mostra e permette di cambiarla (da 10 a 64
# caratteri, senza spazi ne' %). Credenziali Lyrion gia' presenti non vengono
# sovrascritte: scrivile qui. Togli LYRION_PASSWORD per ruotarla.
LYRION_UTENTE=
LYRION_PASSWORD=

# Ottimizzazioni di sistema (core dedicati, frequenza fissa, ecc.).
# no = funzionamento normale, utile solo per confronti o problemi.
OTTIMIZZAZIONI=si

# Watchdog: se il sistema si blocca del tutto il computer riparte da solo
# (dopo un aggiornamento che si blocca, con la versione precedente).
WATCHDOG=si

# Fonte degli aggiornamenti (vuoto = questo progetto su GitHub).
AGGIORNAMENTI_URL=
